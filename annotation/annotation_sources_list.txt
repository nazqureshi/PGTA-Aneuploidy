2. ANNOVAR
    1. annovar script: 
        - script code dropdown
            
            ```r
            #!/usr/bin/bash
            
            # This script is used to generate an annovar annotation file from a vcf file
            
            # hardcode parameters and files directly 
            # Added avsnp151 protocol to get rsid assignments to compare with manually done dbsnp assignment afterwards
            
            # Usage: ./annotate_ANNOVAR_script.sh 
            
            # Set variables based on user inputs
            # Lab-server path roots -- update these to match your own environment.
            PROJECT_ROOT="/path/to/lab_server/aneuploidyProj"
            TOOLS_ROOT="/path/to/lab_server/tools/ANNOVAR/annovar_202006"

            INPUT_VCF="${PROJECT_ROOT}/third_annotatoin_YES_total_embryos/files_to_run_annovar/filtered_nogenotypes.vcf"
            OUTPUT_PREFIX="${PROJECT_ROOT}/third_annotatoin_YES_total_embryos/files_to_run_annovar/annovar_annotated_sig_vars_less_min_5"
            THREADS=12
            
            # Paths to ANNOVAR tools and databases
            ANNOVAR_TOOL="${TOOLS_ROOT}/table_annovar.pl"
            HUMAN_DB="${TOOLS_ROOT}/humandb"
            BUILDVER="hg38"
            PROTOCOL="refGene,ensGene,knownGene,gnomad211_exome,gnomad312_genome,gnomad41_exome,gnomad41_genome,clinvar_20221231,dbnsfp42a,avsnp151"
            OPERATION="g,g,g,f,f,f,f,f,f,f"
            
            perl $ANNOVAR_TOOL \
                $INPUT_VCF \
                $HUMAN_DB \
                -buildver $BUILDVER \
                -otherinfo \
                -protocol $PROTOCOL \
                -operation $OPERATION \
                -nastring "." \
                -vcfinput \
                --thread $THREADS \
                --polish \
                --outfile $OUTPUT_PREFIX \
                2> $OUTPUT_PREFIX.filtered.log
            ```
